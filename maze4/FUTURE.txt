2. service-worker.js
javascript

Copy
const CACHE_NAME = 'maze-game-v1';
const ASSETS_TO_CACHE = [
  '/',
  '/maze4.html',
  '/maze4.js',
  '/icon-192.png',
  '/icon-512.png',
  '/manifest.json'
];

// Install service worker and cache all assets
self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => {
        return cache.addAll(ASSETS_TO_CACHE);
      })
  );
});

// Activate and clean up old caches
self.addEventListener('activate', event => {
  event.waitUntil(
    caches.keys().then(cacheNames => {
      return Promise.all(
        cacheNames.filter(cacheName => {
          return cacheName !== CACHE_NAME;
        }).map(cacheName => {
          return caches.delete(cacheName);
        })
      );
    })
  );
});

// Serve cached content when offline
self.addEventListener('fetch', event => {
  event.respondWith(
    caches.match(event.request)
      .then(response => {
        // Cache hit - return the response from the cached version
        if (response) {
          return response;
        }
        
        // Not in cache - fetch from network
        return fetch(event.request).then(
          networkResponse => {
            // Check if we received a valid response
            if (!networkResponse || networkResponse.status !== 200 || networkResponse.type !== 'basic') {
              return networkResponse;
            }

            // Clone the response
            const responseToCache = networkResponse.clone();

            caches.open(CACHE_NAME)
              .then(cache => {
                cache.put(event.request, responseToCache);
              });

            return networkResponse;
          }
        );
      })
  );
});

Mobile-Friendly Adjustments
You might also want to make sure your game controls work well on mobile. If your game currently uses keyboard controls, consider adding touch controls for mobile users:

javascript

Copy
// Add to maze4.js - Example touch controls
document.addEventListener('DOMContentLoaded', function() {
  // Add touch buttons if needed
  const gameArea = document.getElementById('gameArea'); // Change to your game's container
  
  // Create touchable directional buttons
  const createTouchControls = () => {
    const touchControls = document.createElement('div');
    touchControls.id = 'touchControls';
    touchControls.style.position = 'fixed';
    touchControls.style.bottom = '20px';
    touchControls.style.left = '50%';
    touchControls.style.transform = 'translateX(-50%)';
    touchControls.style.display = 'grid';
    touchControls.style.gridTemplateAreas = '"up up" "left right" "down down"';
    touchControls.style.gap = '10px';
    
    const createButton = (direction, symbol, gridArea) => {
      const button = document.createElement('button');
      button.textContent = symbol;
      button.style.gridArea = gridArea;
      button.style.width = '60px';
      button.style.height = '60px';
      button.style.fontSize = '24px';
      button.style.borderRadius = '10px';
      button.style.backgroundColor = 'rgba(255, 255, 255, 0.7)';
      button.style.border = 'none';
      
      button.addEventListener('touchstart', (e) => {
        e.preventDefault();
        // Simulate key press - update to match your game's key handlers
        const keyEvent = new KeyboardEvent('keydown', { key: direction });
        document.dispatchEvent(keyEvent);
      });
      
      return button;
    };
    
    touchControls.appendChild(createButton('ArrowUp', '↑', 'up'));
    touchControls.appendChild(createButton('ArrowLeft', '←', 'left'));
    touchControls.appendChild(createButton('ArrowRight', '→', 'right'));
    touchControls.appendChild(createButton('ArrowDown', '↓', 'down'));
    
    document.body.appendChild(touchControls);
  };
  
  // Only show touch controls on mobile devices
  if ('ontouchstart' in window || navigator.maxTouchPoints > 0) {
    createTouchControls();
  }
});

How to fix or minimize it

1. Use transform: scale() carefully
If you must scale, scale the entire canvas, not small sub-elements individually.

Example:

html, body {
  transform: scale(1.2);
  transform-origin: top left;
  width: 83.3333%; /* 100% / 1.2 */
  height: 83.3333%;
  overflow: hidden;
}
Adjust width/height inversely so the page layout still fits neatly.

2. Force pixel snapping
You can try to help the browser snap everything to full pixels:

* {
  image-rendering: pixelated;
  transform-style: preserve-3d;
  backface-visibility: hidden;
}
image-rendering: pixelated helps raster images.
preserve-3d and backface-visibility: hidden sometimes trigger cleaner compositing.

3. Use will-change to hint browser compositing
Tell the browser ahead of time that transforms will happen:

body {
  will-change: transform;
}
